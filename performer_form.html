// ============================================================ // 縁日政宗公まつり 奉納演舞者情報フォーム — Google Apps Script // ============================================================ const FOLDER_NAME = '縁日政宗公まつり_演舞者画像'; function doPost(e) { try { const data = JSON.parse(e.postData.contents); const sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); if (sheet.getLastRow() === 0) { sheet.appendRow([ '受信日時', '団体様名', '紹介文・コメント', '駐車台数', 'チラシ部数', 'チラシ受取方法', '郵送先住所', '画像1URL', '画像2URL', '画像3URL', '動画URL' ]); sheet.getRange(1, 1, 1, 11).setFontWeight('bold'); } const imageUrls = []; if (data.images && data.images.length > 0) { const folder = getOrCreateFolder(FOLDER_NAME); for (const img of data.images) { const blob = Utilities.newBlob( Utilities.base64Decode(img.data), img.type, img.name ); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); imageUrls.push(file.getUrl()); } } let videoUrl = ''; if (data.video && data.video.data) { const folder = getOrCreateFolder(FOLDER_NAME); const blob = Utilities.newBlob( Utilities.base64Decode(data.video.data), data.video.type, data.video.name ); const file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); videoUrl = file.getUrl(); } sheet.appendRow([ data.submitted_at, data.group_name, data.comment, data.car_count, data.flyer_count, data.flyer_receive, data.postal_address, imageUrls[0] || '', imageUrls[1] || '', imageUrls[2] || '', videoUrl ]); return ContentService .createTextOutput(JSON.stringify({ status: 'ok' })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService .createTextOutput(JSON.stringify({ status: 'error', message: err.message })) .setMimeType(ContentService.MimeType.JSON); } } function getOrCreateFolder(name) { const folders = DriveApp.getFoldersByName(name); if (folders.hasNext()) return folders.next(); return DriveApp.createFolder(name); }
